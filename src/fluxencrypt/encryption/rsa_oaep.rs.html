1//! RSA-OAEP asymmetric encryption implementation.
2//!
3//! This module provides RSA-OAEP encryption and decryption functionality
4//! for encrypting small amounts of data, typically AES keys in hybrid encryption.
5
6use crate::error::{FluxError, Result};
7use crate::keys::{PrivateKey, PublicKey};
8use aws_lc_rs::rsa::{OAEP_SHA512_MGF1SHA512, OaepPrivateDecryptingKey, PrivateDecryptingKey};
9use rsa::Oaep;
10use rsa::pkcs8::EncodePrivateKey;
11use rsa::sha2::Sha512;
12use zeroize::Zeroizing;

14/// RSA-OAEP cipher for asymmetric encryption operations
15#[derive(Debug)]
16pub struct RsaOaepCipher;
17
18impl RsaOaepCipher {
19    /// Create a new RSA-OAEP cipher
20    pub fn new() -> Self {
21        Self
22    }
23
24    /// Encrypt data with RSA-OAEP
25    ///
26    /// # Arguments
27    /// * `public_key` - The RSA public key to encrypt with
28    /// * `plaintext` - The data to encrypt (must be small enough for RSA)
29    ///
30    /// # Returns
31    /// The encrypted data
32    pub fn encrypt(&self, public_key: &PublicKey, plaintext: &[u8]) -> Result<Vec<u8>> {
33        // Check plaintext size constraints
34        let max_plaintext_len = self.max_plaintext_length(public_key)?;
35        if plaintext.len() > max_plaintext_len {
36            return Err(FluxError::invalid_input(format!(
37                "Plaintext too long for RSA encryption: {} > {}",
38                plaintext.len(),
39                max_plaintext_len
40            )));
41        }
42
43        // Create OAEP padding with SHA-512
44        let padding = Oaep::new::<Sha512>();
45
46        // Get a random number generator
47        let mut rng = rsa::rand_core::OsRng;
48
49        // Encrypt using RSA-OAEP
50        public_key
51            .inner()
52            .encrypt(&mut rng, padding, plaintext)
53            .map_err(|e| FluxError::crypto(format!("RSA encryption failed: {}", e)))
54    }
55
56    /// Decrypt data with RSA-OAEP
57    ///
58    /// # Arguments
59    /// * `private_key` - The RSA private key to decrypt with
60    /// * `ciphertext` - The encrypted data
61    ///
62    /// # Returns
63    /// The decrypted plaintext
64    pub fn decrypt(&self, private_key: &PrivateKey, ciphertext: &[u8]) -> Result<Vec<u8>> {
65        let expected_size = private_key.key_size_bits() / 8;
66        if ciphertext.len() != expected_size {
67            return Err(FluxError::invalid_input(format!(
68                "Invalid ciphertext length: {} != {}",
69                ciphertext.len(),
70                expected_size
71            )));
72        }
73
74        // RustCrypto RSA private operations are affected by RUSTSEC-2023-0071.
75        // Keep its public key/codec API, but perform private operations in AWS-LC.
76        // SecretDocument zeroizes the temporary PKCS#8 encoding on drop.
77        let der = private_key
78            .inner()
79            .to_pkcs8_der()
80            .map_err(|_| FluxError::crypto("RSA decryption failed"))?;
81        let key = PrivateDecryptingKey::from_pkcs8(der.as_bytes())
82            .map_err(|_| FluxError::crypto("RSA decryption failed"))?;
83        let key = OaepPrivateDecryptingKey::new(key)
84            .map_err(|_| FluxError::crypto("RSA decryption failed"))?;
85        let mut plaintext = Zeroizing::new(vec![0; key.min_output_size()]);
86        key.decrypt(&OAEP_SHA512_MGF1SHA512, ciphertext, &mut plaintext, None)
87            .map(|plaintext| plaintext.to_vec())
88            .map_err(|_| FluxError::crypto("RSA decryption failed"))
89    }
90
91    /// Calculate the maximum plaintext length for RSA-OAEP encryption
92    ///
93    /// For RSA-OAEP with SHA-512, the maximum plaintext length is:
94    /// key_length_bytes - 2 * hash_length_bytes - 2
95    /// where hash_length_bytes = 64 for SHA-512
96    pub fn max_plaintext_length(&self, public_key: &PublicKey) -> Result<usize> {
97        let key_size_bytes = public_key.key_size_bits() / 8;
98
99        // For RSA-OAEP with SHA-512: overhead = 2 * 64 + 2 = 130 bytes
100        let oaep_overhead = 130;
101
102        if key_size_bytes <= oaep_overhead {
103            return Err(FluxError::key("RSA key too small for OAEP encryption"));
104        }
105
106        Ok(key_size_bytes - oaep_overhead)
107    }
108
109    /// Get the ciphertext length for a given RSA key
110    pub fn ciphertext_length(&self, public_key: &PublicKey) -> usize {
111        public_key.key_size_bits() / 8
112    }
113}

127    #[test]
128    fn test_aws_lc_sha512_interoperability_and_digest_binding() {
129        use aws_lc_rs::rsa::{OAEP_SHA256_MGF1SHA256, OaepPublicEncryptingKey};
130
131        let keypair = KeyPair::generate(2048).unwrap();
132        let der = keypair.private_key().inner().to_pkcs8_der().unwrap();
133        let native = PrivateDecryptingKey::from_pkcs8(der.as_bytes()).unwrap();
134        let native_public = OaepPublicEncryptingKey::new(native.public_key()).unwrap();
135        let plaintext = b"existing OAEP SHA-512 wire format";
136        let cipher = RsaOaepCipher::new();
137        let mut ciphertext = vec![0; native_public.ciphertext_size()];
138        let encrypted = native_public
139            .encrypt(&OAEP_SHA512_MGF1SHA512, plaintext, &mut ciphertext, None)
140            .unwrap();
141        assert_eq!(
142            cipher.decrypt(keypair.private_key(), encrypted).unwrap(),
143            plaintext
144        );
145
146        // Public encryption remains compatible with the previous RustCrypto path.
147        let legacy_ciphertext = cipher.encrypt(keypair.public_key(), plaintext).unwrap();
148        let native_private = OaepPrivateDecryptingKey::new(native).unwrap();
149        let mut output = Zeroizing::new(vec![0; native_private.min_output_size()]);
150        assert_eq!(
151            native_private
152                .decrypt(
153                    &OAEP_SHA512_MGF1SHA512,
154                    &legacy_ciphertext,
155                    &mut output,
156                    None
157                )
158                .unwrap(),
159            plaintext
160        );
161
162        // Do not silently change the existing OAEP/MGF digest to SHA-256.
163        let wrong_digest = native_public
164            .encrypt(&OAEP_SHA256_MGF1SHA256, plaintext, &mut ciphertext, None)
165            .unwrap();
166        let error = cipher
167            .decrypt(keypair.private_key(), wrong_digest)
168            .unwrap_err();
169        assert_eq!(
170            error.to_string(),
171            FluxError::crypto("RSA decryption failed").to_string()
172        );
173    }

177        let keypair = KeyPair::generate(2048).unwrap();
178        let cipher = RsaOaepCipher::new();
179        let valid = cipher

185        for ciphertext in malformed {
